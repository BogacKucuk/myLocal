#!/usr/bin/env python3
"""
codepack v2 - Uzun kod bloklarini sikistirip tek parca ASCII metne cevirir.
Limitli yerler (MS Teams vb.) icin otomatik parcalara boler.

Varsayilanlar (script'in bulundugu klasorde):
    girdi : raw.txt
    cikti : result.txt
Komut satirinda dosya adi / -o verirsen varsayilanlar ezilir.
Girdi icin "-" stdin, cikti icin "-o -" stdout demektir.

Kullanim:
    python codepack.py pack                       # raw.txt -> result.txt (sikistir + parcala)
    python codepack.py pack Foo.java              # Foo.java -> result.txt
    python codepack.py pack Foo.java -o paket.txt # Foo.java -> paket.txt
    python codepack.py pack -s 12000              # parca boyutunu degistir (varsayilan 20000)
    python codepack.py pack -s 0                  # bolme, tek parca
    python codepack.py pack -d cikti/             # her parca ayri dosyaya
    python codepack.py pack --strip               # satir sonu bosluklarini at (daha kucuk)
    python codepack.py pack --fast                # parametre aramasini atla (hizli)

    python codepack.py unpack                     # raw.txt -> result.txt (parcalari birlestirip coz)
    python codepack.py unpack paket.txt -o Foo.java
    python codepack.py unpack p1.txt p2.txt       # birden fazla dosya (sirasi onemli degil)
    python codepack.py unpack - -o -              # stdin -> stdout

Parcalar bolunmus ise: hepsini ayni dosyaya alt alta yapistir (ya da ayri
dosyalar olarak ver). Sira karisik olabilir, her parcanin CRC'si kontrol
edilir, eksik/bozuk parca varsa hangisi oldugu soylenir.
"""

import argparse
import base64
import hashlib
import lzma
import re
import sys
import zlib
from pathlib import Path

FOOTER = "==CODEPACK-END=="
DEFAULT_CHUNK = 20000      # Teams mesaj limiti ~28 KB; guvenli pay birakildi
LINE_WIDTH = 200           # satir sayisi az olsun (Teams her satiri <br>/<p> yapar)
DEFAULT_INPUT = "raw.txt"
DEFAULT_OUTPUT = "result.txt"

CHUNK_RE = re.compile(
    r"==CODEPACK2 (\d+)/(\d+) id=([0-9a-f]{8}) crc=([0-9a-f]{8})==(.*?)==CODEPACK-END==",
    re.S,
)


class Incomplete(ValueError):
    def __init__(self, missing, bad, total):
        self.missing, self.bad, self.total = missing, bad, total
        msg = f"Eksik parca: {missing} (toplam {total})"
        if bad:
            msg += f"; bozuk (CRC hatali) parca: {sorted(bad)}"
        super().__init__(msg)


# --------------------------------------------------------------------------- #
# Sikistirma
# --------------------------------------------------------------------------- #

def _compress(raw: bytes, fast: bool):
    """LZMA (.lzma / FORMAT_ALONE) - xz kabindan ~50 bayt daha hafif.
    lc/pb parametreleri icin kucuk bir arama yapip en kucugunu secer."""
    dict_size = min(1 << 26, max(1 << 12, 1 << (len(raw) - 1).bit_length()))
    grid = [(3, 2)] if fast else [(lc, pb) for lc in (0, 2, 3, 4) for pb in (0, 2)]
    best = None
    for lc, pb in grid:
        filters = [{
            "id": lzma.FILTER_LZMA1,
            "preset": 9 | lzma.PRESET_EXTREME,
            "dict_size": dict_size,
            "lc": lc, "lp": 0, "pb": pb,
        }]
        try:
            comp = lzma.compress(raw, format=lzma.FORMAT_ALONE, filters=filters)
        except lzma.LZMAError:
            continue
        if best is None or len(comp) < len(best[0]):
            best = (comp, lc, pb)
    return best


def _make_chunks(b64: str, size: int):
    ident = hashlib.sha256(b64.encode("ascii")).hexdigest()[:8]
    if size and size > 0:
        body_max = max(400, int((size - 100) * LINE_WIDTH / (LINE_WIDTH + 1)))
        parts = [b64[i:i + body_max] for i in range(0, len(b64), body_max)]
    else:
        parts = [b64]
    total = len(parts)
    out = []
    for i, p in enumerate(parts, 1):
        crc = f"{zlib.crc32(p.encode('ascii')):08x}"
        body = "\n".join(p[j:j + LINE_WIDTH] for j in range(0, len(p), LINE_WIDTH))
        out.append(f"==CODEPACK2 {i}/{total} id={ident} crc={crc}==\n{body}\n{FOOTER}")
    return out


def pack(text: str, size: int = DEFAULT_CHUNK, strip: bool = False,
         fast: bool = False, info=None):
    if strip:
        text = "\n".join(l.rstrip() for l in text.replace("\r\n", "\n").split("\n"))
    raw = text.encode("utf-8")
    if not raw:
        raise ValueError("Girdi bos.")
    comp, lc, pb = _compress(raw, fast)
    if lzma.decompress(comp) != raw:           # emniyet: gidis-donus dogrulama
        raise RuntimeError("Sikistirma dogrulamasi basarisiz.")
    b64 = base64.b64encode(comp).decode("ascii").rstrip("=")
    chunks = _make_chunks(b64, size)
    if info:
        info(f"[i] {len(raw)} bayt -> {len(comp)} bayt sikistirildi "
             f"(lzma lc={lc} pb={pb}), metin: {len(b64)} kr")
    return chunks


# --------------------------------------------------------------------------- #
# Cozme
# --------------------------------------------------------------------------- #

def _legacy_clean(text: str) -> str:
    """CODEPACK1 (eski surum) ve parca etiketli eski cikti icin."""
    keep = []
    for line in text.splitlines():
        s = line.strip()
        if not s or s.startswith("```") or s.startswith("--") or "CODEPACK" in s:
            continue
        keep.append(s)
    return re.sub(r"[^A-Za-z0-9+/=]", "", "".join(keep))


def _gather(text: str):
    sets = {}
    for m in CHUNK_RE.finditer(text):
        idx, total, ident, crc, body = int(m[1]), int(m[2]), m[3], m[4], m[5]
        body = re.sub(r"[^A-Za-z0-9+/]", "", body)
        s = sets.setdefault(ident, {"total": total, "parts": {}, "bad": set()})
        if f"{zlib.crc32(body.encode('ascii')):08x}" == crc:
            s["parts"][idx] = body
            s["bad"].discard(idx)
        elif idx not in s["parts"]:
            s["bad"].add(idx)
    return sets


def unpack(text: str) -> str:
    sets = _gather(text)
    if sets:
        if len(sets) > 1:
            raise ValueError(
                "Birden fazla farkli paketin parcalari karismis: " + ", ".join(sets))
        s = next(iter(sets.values()))
        total = s["total"]
        missing = [i for i in range(1, total + 1) if i not in s["parts"]]
        if missing:
            raise Incomplete(missing, s["bad"], total)
        b64 = "".join(s["parts"][i] for i in range(1, total + 1))
    else:
        b64 = _legacy_clean(text)
        if not b64:
            raise ValueError("Girdide sikistirilmis veri bulunamadi.")
    b64 += "=" * ((-len(b64)) % 4)
    comp = base64.b64decode(b64)
    return lzma.decompress(comp).decode("utf-8")   # .xz ve .lzma otomatik taninir


# --------------------------------------------------------------------------- #
# CLI
# --------------------------------------------------------------------------- #

def _err(msg):
    print(msg, file=sys.stderr)


def main():
    for stream in (sys.stdin, sys.stdout, sys.stderr):
        try:
            stream.reconfigure(encoding="utf-8")
        except Exception:
            pass

    ap = argparse.ArgumentParser(description="Kod bloklarini sikistir / coz")
    ap.add_argument("mode", choices=["pack", "unpack"])
    ap.add_argument("infile", nargs="*",
                    help=f"Girdi dosya(lari). Varsayilan: {DEFAULT_INPUT}. '-' = stdin")
    ap.add_argument("-o", "--out",
                    help=f"Cikti dosyasi. Varsayilan: {DEFAULT_OUTPUT}. '-' = stdout")
    ap.add_argument("-s", "--split", type=int, default=DEFAULT_CHUNK,
                    help=f"Parca boyutu (karakter). 0 = bolme. Varsayilan {DEFAULT_CHUNK}")
    ap.add_argument("-d", "--dir", help="pack: her parcayi ayri dosya olarak bu klasore yaz")
    ap.add_argument("--strip", action="store_true",
                    help="pack: satir sonu bosluklarini ve CRLF'yi temizle (hafif kayipli)")
    ap.add_argument("--fast", action="store_true",
                    help="pack: en iyi lzma parametresini aramadan hizli sikistir")
    args = ap.parse_args()

    base = Path(__file__).resolve().parent
    files = args.infile or [str(base / DEFAULT_INPUT)]
    out_path = args.out or str(base / DEFAULT_OUTPUT)

    parts = []
    for fn in files:
        if fn == "-":
            parts.append(sys.stdin.read())
            continue
        try:
            with open(fn, "r", encoding="utf-8", newline="") as f:
                parts.append(f.read())
        except FileNotFoundError:
            sys.exit(f"[!] Girdi dosyasi bulunamadi: {fn}")
        _err(f"[i] okundu: {fn}")
    data = "\n".join(parts)

    # ----------------------------- PACK ----------------------------------- #
    if args.mode == "pack":
        chunks = pack(data, args.split, args.strip, args.fast, info=_err)
        n = len(chunks)
        _err(f"[i] {len(data)} kr -> {sum(map(len, chunks))} kr "
             f"(x{len(data) / max(sum(map(len, chunks)), 1):.1f} kucuk), {n} parca, "
             f"en buyuk parca {max(map(len, chunks))} kr")

        if args.dir:
            d = Path(args.dir)
            d.mkdir(parents=True, exist_ok=True)
            stem = Path(files[0]).stem if files[0] != "-" else "codepack"
            for i, c in enumerate(chunks, 1):
                p = d / f"{stem}.part{i}of{n}.txt"
                p.write_text(c + "\n", encoding="utf-8")
                _err(f"[i] yazildi: {p}")
            return

        result = "\n\n".join(chunks)
        if out_path == "-":
            print(result)
        else:
            with open(out_path, "w", encoding="utf-8", newline="") as f:
                f.write(result + "\n")
            _err(f"[i] yazildi: {out_path}")
        return

    # ---------------------------- UNPACK ---------------------------------- #
    try:
        result = unpack(data)
    except ValueError as e:          # Incomplete de ValueError'dan turer
        sys.exit(f"[!] {e}")
    except (lzma.LZMAError, base64.binascii.Error) as e:
        sys.exit(f"[!] Veri bozuk, cozulemedi: {e}")

    _err(f"[i] cozuldu: {len(result)} kr, {result.count(chr(10)) + 1} satir")
    if out_path == "-":
        sys.stdout.write(result)
    else:
        with open(out_path, "w", encoding="utf-8", newline="") as f:
            f.write(result)
        _err(f"[i] yazildi: {out_path}")


if __name__ == "__main__":
    main()
